Flyin is a modernized popup library for creating normal or modal popups with custom positioning, popup open/close animation effect, customized overlay and popup looks, tons of settings. And it's responsive.

Library has over 20 animation effects you can assign, with additional 'random' effect that will pick random effect on every page load.

Example #04: This example doesn't have auto open/close active, and it is not modal popup with overal, and ESC is not closing the popup. But, you have buttons to open/close the popup and to show it's status based on the callback functions.

|

<script type="text/javascript">
  document.addEventListener('DOMContentLoaded', function() {
    const popup = new Flyin("#flyin-example-demo-04", {
      callbacks: {
        beforeOpen: function(obj){
          document.querySelector("#span-event").innerHTML = "beforeOpen";
          document.querySelector("#span-status").innerHTML = obj.get("status");
        },
        afterOpen: function(obj){
          document.querySelector("#span-event").innerHTML = "afterOpen";
          document.querySelector("#span-status").innerHTML = obj.get("status");
        },
        beforeClose: function(obj){
          document.querySelector("#span-event").innerHTML = "beforeClose";
          document.querySelector("#span-status").innerHTML = obj.get("status");
        },
        afterClose: function(obj){
          document.querySelector("#span-event").innerHTML = "afterClose";
          document.querySelector("#span-status").innerHTML = obj.get("status");
        }
      },
      settings: {
        modal: false,
        closeEscape: false,
        onLoad: false,
        footer: false,
        positionX: "right",
        positionY: "bottom",
        width: "30%",
        minWidth: "320px"
      }
    });

    document.querySelector("#snap-open-popup").addEventListener('click', function(){
      popup.open();
    });

    document.querySelector("#snap-close-popup").addEventListener('click', function(){
      popup.close();
    });
  });
</script>
